# Fix House of York Founding
#
# Issue: flavor_eng.45 only *weights* its candidate toward being a son of
# Edward III instead of requiring it, so at the 1337 start it almost always
# hands York to eng_thomas_of_brotherton - the only other eligible male
# Plantagenet, and a dead end with no living son. The dynasty dies with him.
#
# Fix: require (not just weight) the candidate to be a descendant of Edward
# III via any_ancestor, so the event can't fire until he has an adult son -
# matching the real founder, Edmund of Langley - and keeps working for later
# generations too.
#
# Note on how this override loads: events aren't part of the merge-aware
# INJECT:/REPLACE: system (unlike laws/gods/modifiers), so this redefines
# flavor_eng.45 under its bare ID. EU5 keeps the FIRST-loaded definition of a
# given event ID, so this file is named `0000_...` to sort before vanilla's
# `flavor_ENG.txt` and load first - which is also why it declares its own
# `namespace = flavor_eng` instead of relying on vanilla's declaration.

namespace = flavor_eng

flavor_eng.45 = {
	type = country_event

	fire_only_once = yes

	title = flavor_eng.45.title
	desc = flavor_eng.45.desc
	outcome = neutral
	historical_info = flavor_eng.45.historical_info
	dynamic_historical_event = {
		tag = ENG
		from = 1300.1.1
		to = 1500.1.1
		monthly_chance = 33
	}


	trigger = {
		has_ruler = yes
		owns_or_non_sovereign_subject_owns = location:york
		ruler.dynasty ?= dynasty:plantagenet_dynasty
		dynasty:plantagenet_dynasty = {
			any_character_in_dynasty = {
				is_heir = no
				is_ruler = no
				is_female = no
				is_adult = yes
				# HOY fix: only a descendant of Edward III qualifies.
				any_ancestor = {
					custom_description = {
						text = hoy_is_edward_iii_descendant
						this = character:eng_edward_iii
					}
				}
			}
		}

		dynasty:lancaster_dynasty ?= {
			any_character_in_dynasty = {
				count > 0
			}
		}
	}

	illustration_tags = {
        10 = regular
        10 = interior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		dynasty:plantagenet_dynasty = {
			random_character_in_dynasty = {
				limit = {
					is_heir = no
					is_ruler = no
					is_female = no
					is_adult = yes
					# HOY fix: only a descendant of Edward III qualifies.
					any_ancestor = {
						custom_description = {
							text = hoy_is_edward_iii_descendant
							this = character:eng_edward_iii
						}
					}
				}
				weight = {
					base = 1
				}
				save_scope_as = target_character
				change_dynasty = dynasty:york_dynasty
				every_descendant = {
					change_dynasty = dynasty:york_dynasty
				}
			}
			scope:target_character.dynasty = { save_scope_as = target_dynasty }
		}
	}

	option = {
		name = flavor_eng.45.a
		show_as_tooltip = {
			scope:target_character = {
				change_dynasty = dynasty:york_dynasty
				every_descendant = {
					change_dynasty = dynasty:york_dynasty
				}
			}
		}
		add_prestige = prestige_extreme_bonus
	}

	# HOY fix: vanilla never scheduled flavor_eng.666 (spouse backstop) after
	# York's founding, only Lancaster's - add the same hook here.
	after = {
		trigger_event_silently = {
			id = flavor_eng.666
			months = { 400 500 }
		}
	}
}

# Fix Founding of House Lancaster
#
# Issue: flavor_eng.46 has the same flaw as flavor_eng.45 - it only weights
# toward a son of Edward III rather than requiring it. It doesn't die out
# immediately (it force-marries its pick and grants +100 fertility), but it
# can still crown the wrong historical founder.
#
# Fix: same as flavor_eng.45 - require an any_ancestor descendant of Edward
# III. Everything else is untouched vanilla behaviour.

flavor_eng.46 = {
	type = country_event

	fire_only_once = yes

	title = flavor_eng.46.title
	desc = flavor_eng.46.desc
	outcome = neutral
	historical_info = flavor_eng.46.historical_info

	dynamic_historical_event = {
		tag = ENG
		from = 1300.1.1
		to = 1500.1.1
		monthly_chance = 33
	}


	trigger = {
		has_ruler = yes
		owns_or_non_sovereign_subject_owns = location:lancaster
		ruler.dynasty ?= dynasty:plantagenet_dynasty
		dynasty:plantagenet_dynasty = {
			any_character_in_dynasty = {
				is_heir = no
				is_ruler = no
				is_female = no
				is_adult = yes
				is_married = no
				age_in_years < 40
				is_alive = yes
				# HOY fix: only a descendant of Edward III qualifies.
				any_ancestor = {
					custom_description = {
						text = hoy_is_edward_iii_descendant
						this = character:eng_edward_iii
					}
				}
			}
		}
		dynasty:lancaster_dynasty ?= {
			any_character_in_dynasty = {
				is_heir = no
				is_ruler = no
				is_female = yes
				is_adult = yes
				is_married = no
				age_in_years < 35
				is_alive = yes
			}
		}
	}

	illustration_tags = {
        10 = happy
        10 = interior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		dynasty:plantagenet_dynasty = {
			random_character_in_dynasty = {
				limit = {
					is_heir = no
					is_ruler = no
					is_female = no
					is_adult = yes
					is_married = no
					age_in_years < 40
					is_alive = yes
					# HOY fix: only a descendant of Edward III qualifies.
					any_ancestor = {
						custom_description = {
							text = hoy_is_edward_iii_descendant
							this = character:eng_edward_iii
						}
					}
				}
				weight = {
					base = 1
					modifier = {
						factor = 100
						trigger = {
							exists = father
							father ?= character:eng_edward_iii
						}
					}
				}
				add_fertility = 100
				save_scope_as = target
			}
		}
		dynasty:lancaster_dynasty ?= {
			random_character_in_dynasty = {
				limit = {
					is_heir = no
					is_ruler = no
					is_female = yes
					is_adult = yes
					is_married = no
					age_in_years < 35
					is_alive = yes
				}
				add_fertility = 100
				save_scope_as = target_female
			}
		}
	}

	option = {
		name = flavor_eng.46.a
		scope:target = {
			change_dynasty = dynasty:lancaster_dynasty
			marry_character = scope:target_female

			every_descendant = {
				change_dynasty = dynasty:lancaster_dynasty
			}
		}
		add_prestige = prestige_extreme_bonus
	}

	after = {
		trigger_event_silently = {
			id = flavor_eng.666
			months = { 400 500 }
		}
	}
}

# Guarantee sons for House York/Lancaster
#
# Issue: a dynasty head with no living son is a dead end (same problem
# flavor_eng.47 solves for Edward III), but the founders here are created
# anonymously so there's no fixed id to target. A first attempt used a
# recurring dynamic_historical_event, but the engine requires
# `fire_only_once = yes` on any dynamic_historical_event - without it the
# event is logged as invalid and never actually scheduled (error.log:
# "is tagged as an dynamic historical event, but is not set as
# fire_only_once"), so it silently never fired.
#
# Fix: flavor_eng.667 is a fire_only_once dynamic_historical_event whose only
# job is to kick off flavor_eng.669, a plain hidden event that reschedules
# itself indefinitely (like vanilla's flavor_eng.666) to actually do the
# recurring check. This applies automatically to any save, including ones
# where founding already happened - no console/manual trigger needed, which
# matters on ironman. Runs for the whole campaign, not just until 1500/the
# War of the Roses, unlike flavor_eng.666.
flavor_eng.667 = {
	type = country_event

	title = flavor_eng.46.title
	desc = flavor_eng.46.desc
	outcome = neutral
	hidden = yes
	fire_only_once = yes

	dynamic_historical_event = {
		tag = ENG
		from = 1300.1.1
		to = 1836.12.31
		monthly_chance = 100
	}

	option = {
		name = flavor_eng.46.a
	}

	after = {
		trigger_event_silently = {
			id = flavor_eng.669
			months = { 1 1 }
		}
	}
}

# Recurring checker kicked off once by flavor_eng.667 - see its comment.
#
# Checks each dynasty's *current* dynasty_head for fewer than 2 living sons
# (a buffer, so losing one to disease doesn't leave zero). When true, flags
# that character and schedules flavor_eng.668 ~5 years out to grant two sons
# at once, so it reads as a real pregnancy rather than an instant child
# appearing. Reschedules itself every 1-2 years, forever.
#
# TEMP TESTING TWEAK: delays below shortened to 1 month so this can be
# verified quickly in an existing save - revert with `git checkout HEAD --
# in_game/events/DHE/0000_HOY_flavor_ENG.txt` when done.
flavor_eng.669 = {
	type = country_event

	title = flavor_eng.46.title
	desc = flavor_eng.46.desc
	outcome = neutral
	hidden = yes

	immediate = {
		if = {
			limit = {
				dynasty:lancaster_dynasty ?= {
					dynasty_head ?= {
						is_alive = yes
						any_child = { count < 2 is_alive = yes is_female = no }
						NOT = { has_variable = hoy_awaiting_heir }
					}
				}
			}
			dynasty:lancaster_dynasty = { dynasty_head = { set_variable = hoy_awaiting_heir } }
			trigger_event_silently = { id = flavor_eng.668 months = { 1 1 } }
		}
		if = {
			limit = {
				dynasty:york_dynasty ?= {
					dynasty_head ?= {
						is_alive = yes
						any_child = { count < 2 is_alive = yes is_female = no }
						NOT = { has_variable = hoy_awaiting_heir }
					}
				}
			}
			dynasty:york_dynasty = { dynasty_head = { set_variable = hoy_awaiting_heir } }
			trigger_event_silently = { id = flavor_eng.668 months = { 1 1 } }
		}
	}

	option = {
		name = flavor_eng.46.a
	}

	after = {
		trigger_event_silently = {
			id = flavor_eng.669
			months = { 1 1 }
		}
	}
}

# Payoff half of flavor_eng.669 - see that event's comment.
#
# Finds whichever dynasty member was flagged ~5 years ago and gives them two
# sons at once, then clears the flag.
flavor_eng.668 = {
	type = country_event

	title = flavor_eng.46.title
	desc = flavor_eng.46.desc
	outcome = neutral
	hidden = yes

	immediate = {
		if = {
			limit = {
				dynasty:lancaster_dynasty ?= {
					any_character_in_dynasty = { has_variable = hoy_awaiting_heir is_alive = yes }
				}
			}
			dynasty:lancaster_dynasty = {
				random_character_in_dynasty = {
					limit = { has_variable = hoy_awaiting_heir is_alive = yes }
					save_scope_as = lancaster_father
				}
			}
			create_character = { dynasty = dynasty:lancaster_dynasty birth_location = location:lancaster father = scope:lancaster_father }
			create_character = { dynasty = dynasty:lancaster_dynasty birth_location = location:lancaster father = scope:lancaster_father }
			scope:lancaster_father = { remove_variable = hoy_awaiting_heir }
		}
		if = {
			limit = {
				dynasty:york_dynasty ?= {
					any_character_in_dynasty = { has_variable = hoy_awaiting_heir is_alive = yes }
				}
			}
			dynasty:york_dynasty = {
				random_character_in_dynasty = {
					limit = { has_variable = hoy_awaiting_heir is_alive = yes }
					save_scope_as = york_father
				}
			}
			create_character = { dynasty = dynasty:york_dynasty birth_location = location:york father = scope:york_father }
			create_character = { dynasty = dynasty:york_dynasty birth_location = location:york father = scope:york_father }
			scope:york_father = { remove_variable = hoy_awaiting_heir }
		}
	}

	option = {
		name = flavor_eng.46.a
	}
}
