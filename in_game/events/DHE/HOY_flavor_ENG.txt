# Fix House of York Founding
#
# Vanilla's flavor_eng.45 ("Founding of House of York") picks a random adult,
# non-ruler, non-heir male Plantagenet to found House of York, only *weighting*
# (not requiring) candidates who are sons of Edward III. At the 1337 start,
# none of Edward III's own sons are adults yet, and every other eligible male
# Plantagenet is already dead (Edmund of Woodstock d.1330, Edmund of Kent
# d.1331, Edward of Norfolk d.1334, John of Eltham d.1336) except
# eng_thomas_of_brotherton, who has no living son (his only son died in 1334)
# and nothing in the game reliably arranges his remarriage (marry_noble.txt's
# AI urgency weighting only applies to rulers/heirs). So the event is
# essentially guaranteed to hand York to the wrong historical person, and the
# freshly-founded dynasty dies with him.
#
# Fix: require the candidate to actually be a son of Edward III (matching the
# real historical founder, Edmund of Langley), both in the firing trigger and
# in the random-character selection, instead of merely weighting toward it.
# This means the event can't fire until Edward III has an adult son - exactly
# when it should, historically circa 1360 - rather than firing early on
# whichever unrelated uncle happens to be available.
#
# This REPLACEs the whole event (rather than INJECTing) because the fix edits
# conditions nested inside existing trigger/limit blocks (an any_character_in
# _dynasty existence check and a random_character_in_dynasty limit), not
# fields that can be additively merged - INJECTing a second top-level
# `trigger = {}`/`immediate = {}` block would AND/execute independently
# rather than tightening the same nested character filter. Tradeoff: this
# embeds vanilla content and needs re-syncing if Paradox changes this event.

REPLACE:flavor_eng.45 = {
	type = country_event

	fire_only_once = yes

	title = flavor_eng.45.title
	desc = flavor_eng.45.desc
	outcome = neutral
	historical_info = flavor_eng.45.historical_info
	dynamic_historical_event = {
		tag = ENG
		from = 1300.1.1
		to = 1500.1.1
		monthly_chance = 33
	}


	trigger = {
		has_ruler = yes
		owns_or_non_sovereign_subject_owns = location:york
		ruler.dynasty ?= dynasty:plantagenet_dynasty
		dynasty:plantagenet_dynasty = {
			any_character_in_dynasty = {
				is_heir = no
				is_ruler = no
				is_female = no
				is_adult = yes
				# TE_HOY fix: only a real son of Edward III qualifies.
				exists = father
				father ?= character:eng_edward_iii
			}
		}

		dynasty:lancaster_dynasty ?= {
			any_character_in_dynasty = {
				count > 0
			}
		}
	}

	illustration_tags = {
        10 = regular
        10 = interior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		dynasty:plantagenet_dynasty = {
			random_character_in_dynasty = {
				limit = {
					is_heir = no
					is_ruler = no
					is_female = no
					is_adult = yes
					# TE_HOY fix: only a real son of Edward III qualifies.
					exists = father
					father ?= character:eng_edward_iii
				}
				weight = {
					base = 1
				}
				save_scope_as = target_character
				change_dynasty = dynasty:york_dynasty
				every_descendant = {
					change_dynasty = dynasty:york_dynasty
				}
			}
			scope:target_character.dynasty = { save_scope_as = target_dynasty }
		}
	}

	option = {
		name = flavor_eng.45.a
		show_as_tooltip = {
			scope:target_character = {
				change_dynasty = dynasty:york_dynasty
				every_descendant = {
					change_dynasty = dynasty:york_dynasty
				}
			}
		}
		add_prestige = prestige_extreme_bonus
	}

}
